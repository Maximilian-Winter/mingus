module Test02
{
    extern func printf(string fmt, ...) => int;
    extern func puts(string s) => int;
    extern func sqrt(double x) => double;

    struct Vec3
    {
        double x;
        double y;
        double z;

        func operator+(Vec3 other) => Vec3
        {
            Vec3 result;
            result.x = this.x + other.x;
            result.y = this.y + other.y;
            result.z = this.z + other.z;
            return result;
        }

        func operator*(double scalar) => Vec3
        {
            Vec3 result;
            result.x = this.x * scalar;
            result.y = this.y * scalar;
            result.z = this.z * scalar;
            return result;
        }

        func dot(Vec3 other) => double
        {
            return this.x * other.x + this.y * other.y + this.z * other.z;
        }

        func cross(Vec3 other) => Vec3
        {
            Vec3 result;
            result.x = this.y * other.z - this.z * other.y;
            result.y = this.z * other.x - this.x * other.z;
            result.z = this.x * other.y - this.y * other.x;
            return result;
        }

        func length() => double
        {
            return sqrt(this.dot(this));
        }

        func normalize() => Vec3
        {
            var len = this.length();
            return this * (1.0 / len);
        }
    }

    func main() => int
    {
        puts("=== Test 02: Structs & Operators ===");

        // Create vectors
        Vec3 a;
        a.x = 1.0;
        a.y = 0.0;
        a.z = 0.0;

// Test 3: Classes, constructors, destructors, RAII, new/delete
module Test03
{
    extern func printf(string fmt, ...) => int;
    extern func puts(string s) => int;
    extern func malloc(int size) => byte*;
    extern func free(byte* ptr) => void;
    extern func memcpy(byte* dest, byte* src, int n) => byte*;

    class DynamicArray
    {
        private int* data;
        private int size;
        private int capacity;

        constructor(int initialCapacity)
        {
            this.capacity = initialCapacity;
            this.size = 0;
            raw { this.data = (int*)malloc(initialCapacity * sizeof(int)); }
        }

        destructor
        {
            if (this.data != null)
            {
                raw { free((byte*)this.data); }
            }
            puts("  [DynamicArray destructor called]");
        }

        func push(int value) => void
        {
            if (this.size >= this.capacity)
            {
                this.grow();
            }

            raw
            {
                *(this.data + this.size) = value;
            }

            this.size++;
        }

        func operator[](int index) => int
        {
            raw
            {
                return *(this.data + index);
            }
        }

        func length() => int
        {
            return this.size;
        }

        private func grow() => void
        {
            var newCapacity = this.capacity * 2;

            raw
            {
                var newData = (int*)malloc(newCapacity * sizeof(int));
                memcpy((byte*)newData, (byte*)this.data, this.size * sizeof(int));
                free((byte*)this.data);
                this.data = newData;
            }

            this.capacity = newCapacity;
        }
    }

    class Node
    {
        int value;
        Node* left;
        Node* right;

        constructor(int value)
        {
            this.value = value;
            this.left = null;
            this.right = null;
        }

        destructor
        {
            if (this.left != null)
            {
                delete this.left;
            }
            if (this.right != null)
            {
                delete this.right;
            }
        }
    }

    func sumTree(Node* node) => int
    {
        if (node == null)
        {
            return 0;
        }
        return node->value + sumTree(node->left) + sumTree(node->right);
    }

    func main() => int
    {
        puts("=== Test 03: Classes & RAII ===");

        // Test DynamicArray with constructor + push + operator[]
        puts("Creating DynamicArray(4)...");
        {
            var arr = DynamicArray(4);

            for (int i = 0; i < 10; i++)
            {
                arr.push(i * 10);
            }

            printf("arr.length() = %d (expect 10)\n", arr.length());
            printf("arr[0] = %d (expect 0)\n", arr[0]);
            printf("arr[3] = %d (expect 30)\n", arr[3]);
            printf("arr[9] = %d (expect 90)\n", arr[9]);

            puts("Leaving scope...");
        }
        // Destructor should have been called

        // Test new/delete with tree
        puts("\nBuilding tree: 10(5(2,7),15)...");
        var root = new Node(10);
        root->left = new Node(5);
        root->right = new Node(15);
        root->left->left = new Node(2);
        root->left->right = new Node(7);

        var total = sumTree(root);
        printf("Tree sum = %d (expect 39)\n", total);

        puts("Deleting tree...");
        delete root;

        puts("=== Test 03 complete ===");
        return 0;
    }
}

// Test 4: Pipes, pattern matching with guards, match expressions
module Test04
{
    extern func printf(string fmt, ...) => int;
    extern func puts(string s) => int;

    func applyGain(double sample, double gain) => double
    {
        return sample * gain;
    }

    func softClip(double sample) => double
    {
        return match sample {
            var x if x > 1.0  => 1.0,
            var x if x < -1.0 => -1.0,
            var x => x - (x * x * x) / 3.0,
        };
    }

    func classify(double value) => int
    {
        // Returns a category: 0=silent, 1=quiet, 2=nominal, 3=hot, 4=clipping
        return match value {
            var x if x > 0.9  => 4,
            var x if x > 0.7  => 3,
            var x if x > 0.3  => 2,
            var x if x > 0.01 => 1,
            _ => 0,
        };
    }

    func processSample(double sample) => double
    {
        return sample
            |> applyGain(1.5)
            |> softClip;
    }

    func main() => int
    {
        puts("=== Test 04: Pipes & Pattern Matching ===");

        // Simple pipe
        var x = 0.5;
        var gained = x |> applyGain(2.0);
        printf("0.5 |> applyGain(2.0) = %f (expect 1.0)\n", gained);

        // Chained pipe
        var result = 0.7 |> applyGain(1.5) |> softClip;
        printf("0.7 |> gain(1.5) |> clip = %f (expect 1.0)\n", result);

        // Pipe through processSample
        var p1 = 0.3 |> processSample;
        printf("processSample(0.3) = %f\n", p1);

        // Match: softClip cases
        var c1 = softClip(0.5);
        printf("softClip(0.5) = %f (expect ~0.458)\n", c1);

        var c2 = softClip(2.0);
        printf("softClip(2.0) = %f (expect 1.0)\n", c2);

        var c3 = softClip(-2.0);
        printf("softClip(-2.0) = %f (expect -1.0)\n", c3);

        var c4 = softClip(0.0);
        printf("softClip(0.0) = %f (expect 0.0)\n", c4);

        // Integer match for classification
        printf("\nClassification results:\n", 0.0);
        var cat1 = classify(0.95);
        printf("  classify(0.95) = %f (expect 4=clipping)\n", cat1 * 1.0);
        var cat2 = classify(0.5);
        printf("  classify(0.50) = %f (expect 2=nominal)\n", cat2 * 1.0);
        var cat3 = classify(0.001);
        printf("  classify(0.001) = %f (expect 0=silent)\n", cat3 * 1.0);

        puts("=== Test 04 complete ===");
        return 0;
    }
}

module Test05
{
    extern func printf(string fmt, ...) => int;
    extern func puts(string s) => int;

    enum TokenKind : int
    {
        Number     = 0,
        Identifier = 1,
        Plus       = 2,
        Minus      = 3,
        Star       = 4,
        Slash      = 5,
    }

    func tokenPrecedence(int kind) => int
    {
        return match kind {
            TokenKind.Plus  => 1,
            TokenKind.Minus => 1,
            TokenKind.Star  => 2,
            TokenKind.Slash => 2,
            _ => 0,
        };
    }

    func isOperator(int kind) => bool
    {
        return match kind {
            TokenKind.Plus  => true,
            TokenKind.Minus => true,
            TokenKind.Star  => true,
            TokenKind.Slash => true,
            _ => false,
        };
    }

    func tokenName(int kind) => int
    {
        // Return a code for the token name (match with enum patterns)
        return match kind {
            TokenKind.Number     => 100,
            TokenKind.Identifier => 200,
            TokenKind.Plus       => 300,
            TokenKind.Minus      => 400,
            TokenKind.Star       => 500,
            TokenKind.Slash      => 600,
            _ => 0,
        };
    }

    func main() => int
    {
        puts("=== Test 05: Enums and Switch ===");

        return transform(x);
    }

    {
        return f(f(x));
    }

    {

// Test 11: DSP Showcase — A musical tribute to Charles Mingus
// Weaves together: structs + operators, enums + match, closures, pipes, math
module Test11
{
    extern func printf(string fmt, ...) => int;
    extern func puts(string s) => int;
    extern func sin(double x) => double;
    extern func sqrt(double x) => double;

    // ── Stereo sample: struct + operator overloading ──

    struct Sample
    {
        double left;
        double right;

        func operator+(Sample other) => Sample
        {
            Sample out;
            out.left  = this.left  + other.left;
            out.right = this.right + other.right;
            return out;
        }

        func operator*(double gain) => Sample
        {
            Sample out;
            out.left  = this.left  * gain;
            out.right = this.right * gain;
            return out;
        }

        func rms() => double
        {
            return sqrt((this.left * this.left + this.right * this.right) / 2.0);
        }

        func pan(double p) => Sample
        {
            Sample out;
            out.left  = this.left  * (1.0 - p) / 2.0;
            out.right = this.right * (1.0 + p) / 2.0;
            return out;
        }
    }

    // ── Waveform selection: enum + match ──

    enum Wave : int
    {
        Sine     = 0,
        Square   = 1,
        Triangle = 2,
        Saw      = 3,
    }

    func twoPi() => double { return 6.28318530717959; }

    func oscillator(int wave, double phase) => double
    {
        return match wave {
            Wave.Sine     => sin(phase * twoPi()),
            Wave.Square   => phase < 0.5 ? 1.0 : -1.0,
            Wave.Triangle => phase < 0.5
                ? phase * 4.0 - 1.0
                : 3.0 - phase * 4.0,
            _ => phase * 2.0 - 1.0,
        };
    }

    func stereoOsc(int wave, double phase) => Sample
    {
        var mono = oscillator(wave, phase);
        Sample s;
        s.left  = mono;
        s.right = mono;
        return s;
    }

    // ── Effects: closures + higher-order functions ──

    func makeGain(double amount) => (double) => double
    {
        return [=](double x) => { return x * amount; };
    }

    func softClip(double x) => double
    {
        return match x {
            var v if v >  1.0 =>  1.0,
            var v if v < -1.0 => -1.0,
            var v => v - (v * v * v) / 3.0,
        };
    }

    func makeDrive(double drive) => (double) => double
    {
        return [=](double x) => { return softClip(x * drive); };
    }

    func composeEffect((double) => double f, (double) => double g) => (double) => double
    {
        return [=](double x) => { return f(g(x)); };
    }

    func applyStereo(Sample s, (double) => double fx) => Sample
    {
        Sample out;
        out.left  = fx(s.left);
        out.right = fx(s.right);
        return out;
    }

    func transform(double x, (double) => double f) => double
    {
        return f(x);
    }

    // ── Main: the full signal chain ──

    func main() => int
    {
        puts("=== Test 11: DSP Showcase ===");

        // --- Oscillators (enum + match + math) ---
        puts("--- Oscillators ---");

        printf("square(0.25)   = %f\n", oscillator(Wave.Square, 0.25));
        printf("triangle(0.25) = %f\n", oscillator(Wave.Triangle, 0.25));
        printf("saw(0.75)      = %f\n", oscillator(Wave.Saw, 0.75));

        // --- Stereo mixing (struct + operators) ---
        puts("--- Stereo Mix ---");
        var tone = stereoOsc(Wave.Sine, 0.25);
        var buzz = stereoOsc(Wave.Triangle, 0.375);
        printf("tone = %f\n", tone.left);
        printf("buzz = %f\n", buzz.left);

        printf("mix  = %f\n", mixed.left);

        var panned = tone.pan(1.0);

        printf("chain L = %f\n", processed.left);
        printf("chain R = %f\n", processed.right);

        puts("=== Test 11 complete ===");
        return 0;
    }

  --- Source ---
// Test 16: DSP WAV Synthesis
//
// A synthesizer that writes a real WAV file.
// Demonstrates: interfaces (Effect, Named), single inheritance
// (AudioEffect → GainEffect / DriveEffect / SoftClipEffect),
// class-based oscillator with mutable state, pattern matching,
// and fat-pointer interface dispatch in a real-time render loop.
//
// Output: output.wav — 3 seconds, mono, 44100 Hz, 16-bit PCM
//         A3 sine (220 Hz) + E4 triangle (330 Hz) through
//         a Gain → Drive → SoftClip effect chain.

module DspWav
{
    // ---- C standard library ----------------------------------------------------------------------------------------

    extern func fopen(string path, string mode) => byte*;
    extern func fputc(int c, byte* f) => int;
    extern func fclose(byte* f) => int;
    extern func puts(string s) => int;
    extern func printf(string fmt, ...) => int;
    extern func lround(double x) => int;
    extern func sin(double x) => double;

    // ---- WAV writing primitives --------------------------------------------------------------------------------

    func putU16LE(byte* f, int v) => void
    {
        fputc(v & 255, f);
        fputc((v >> 8) & 255, f);
    }

    func putU32LE(byte* f, int v) => void
    {
        fputc(v & 255, f);
        fputc((v >> 8) & 255, f);
        fputc((v >> 16) & 255, f);
        fputc((v >> 24) & 255, f);
    }

    // Write a 44-byte PCM WAV header (mono, 16-bit).
    // ASCII codes used for chunk IDs to avoid string cast complexity.
    func writeWavHeader(byte* f, int sampleRate, int numSamples) => void
    {
        int dataSize = numSamples * 2;       // 1 channel * 2 bytes per sample
        int riffSize = dataSize + 36;
        int byteRate = sampleRate * 2;

        // "RIFF"
        fputc(82, f); fputc(73, f); fputc(70, f); fputc(70, f);
        putU32LE(f, riffSize);
        // "WAVE"
        fputc(87, f); fputc(65, f); fputc(86, f); fputc(69, f);
        // "fmt "
        fputc(102, f); fputc(109, f); fputc(116, f); fputc(32, f);
        putU32LE(f, 16);        // PCM sub-chunk size
        putU16LE(f, 1);         // PCM audio format
        putU16LE(f, 1);         // mono
        putU32LE(f, sampleRate);
        putU32LE(f, byteRate);
        putU16LE(f, 2);         // block align (1 ch * 2 bytes)
        putU16LE(f, 16);        // bits per sample
        // "data"
        fputc(100, f); fputc(97, f); fputc(116, f); fputc(97, f);
        putU32LE(f, dataSize);
    }

    // Clamp, scale, and write one 16-bit PCM sample (little-endian).
    func writeSample(byte* f, double sample) => void
    {
        var clamped = sample > 1.0 ? 1.0 : (sample < -1.0 ? -1.0 : sample);
        int ival = lround(clamped * 32767.0);
        putU16LE(f, ival);
    }

    // ---- Interfaces --------------------------------------------------------------------------------------------------------

    interface Effect
    {
        func process(double s) => double;
    }

    interface Named
    {
        func name() => string;
    }

    // ---- Abstract base class --------------------------------------------------------------------------------------
    //
    // Holds a single 'param' double used differently by each effect.
    // Cannot be instantiated; subclasses must implement Effect and Named.

    abstract class AudioEffect
    {
        double param;
        constructor(double p) { this.param = p; }
        // A virtual method is required so that AudioEffect gets a vtable pointer
        // in its struct layout — { ptr, double } — matching the layout of every
        // derived class and ensuring super() stores 'param' at the correct offset.
        func paramValue() => double { return this.param; }
    }

    // ---- Concrete effects --------------------------------------------------------------------------------------------

    // GainEffect: simple linear amplitude scaling.
    class GainEffect : AudioEffect, Effect, Named
    {
        constructor(double g) : super(g) { }
        func process(double s) => double { return s * this.param; }
        func name() => string { return "Gain"; }
    }

    // DriveEffect: soft saturation — x - x³/3.
    // param is the drive multiplier applied before the curve.
    class DriveEffect : AudioEffect, Effect, Named
    {
        constructor(double d) : super(d) { }
        func process(double s) => double
        {
            var x = s * this.param;
            return x - (x * x * x) / 3.0;
        }
        func name() => string { return "Drive"; }
    }

    // SoftClipEffect: hard limit at ±threshold with a cubic roll-off inside.
    class SoftClipEffect : AudioEffect, Effect, Named
    {
        constructor(double threshold) : super(threshold) { }
        func process(double s) => double
        {
            var t = this.param;
            return match s {
                var x if x >  t =>  t,
                var x if x < -t => -t,
                var x           => x - (x * x * x) / 3.0,
            };
        }
        func name() => string { return "SoftClip"; }
    }

    // ---- Oscillator --------------------------------------------------------------------------------------------------------
    //
    // Class (reference type) so that phase state can be mutated
    // between tick() calls without passing pointers manually.

    class Oscillator
    {
        double phase;
        double phaseInc;
        int    waveform;    // 0=sine  1=square  2=saw  3=triangle

        constructor(int wave, double freq, double rate)
        {
            this.waveform = wave;
            this.phase    = 0.0;
            this.phaseInc = freq / rate;
        }

        func tick() => double
        {
            var p = this.phase;
            var s = match this.waveform {
                0 => sin(p * 6.283185307),
                1 => p < 0.5 ? 1.0 : -1.0,
                2 => p * 2.0 - 1.0,
                _ => p < 0.5 ? p * 4.0 - 1.0 : 3.0 - p * 4.0,
            };
            var next = p + this.phaseInc;
            this.phase = next >= 1.0 ? next - 1.0 : next;
            return s;
        }
    }

    // ---- Signal chain ----------------------------------------------------------------------------------------------------
    //
    // Three Effect* fat pointers dispatched in sequence.
    // Each call resolves through the class's itable at runtime.

    func applyChain(double s, Effect* e0, Effect* e1, Effect* e2) => double
    {
        var s0 = e0->process(s);
        var s1 = e1->process(s0);
        return e2->process(s1);
    }

    // Print the name of an effect through the Named interface.
    func printEffectName(Named* n) => void
    {
        puts(n->name());
    }

    // ---- Main --------------------------------------------------------------------------------------------------------------------

    func main() => int
    {
        puts("=== DSP WAV Example ===");

        // Construct the effect chain
        GainEffect*     gain  = new GainEffect(0.7);
        DriveEffect*    drive = new DriveEffect(1.8);
        SoftClipEffect* clip  = new SoftClipEffect(0.9);

        // Print effect chain via Named interface
        puts("Effect chain:");
        Named* n0 = gain;
        Named* n1 = drive;
        Named* n2 = clip;
        printEffectName(n0);
        printEffectName(n1);
        printEffectName(n2);

        // Effect interface pointers for the render loop
        Effect* e0 = gain;
        Effect* e1 = drive;

// Test 19: DynamicArray.map with Lambda + Pipe, memcpy for grow
// Tests first-class functions passed to class methods, pipe inside methods,
// and memcpy interop for array resizing.

module Test19
{
    extern func puts(string s) => int;
    extern func printf(string fmt, ...) => int;
    extern func malloc(int size) => byte*;
    extern func free(byte* ptr) => void;
    extern func memcpy(byte* dest, byte* src, int n) => byte*;

    class DynamicArray
    {
        private int* data;
        private int size;
        private int capacity;

        constructor(int initialCapacity)
        {
            this.capacity = initialCapacity;
            this.size = 0;
            raw { this.data = (int*)malloc(initialCapacity * sizeof(int)); }
        }

        destructor
        {
            if (this.data != null)
            {
                raw { free((byte*)this.data); }
            }
        }

        func push(int value) => void
        {
            if (this.size >= this.capacity)
            {
                this.grow();
            }
            raw { *(this.data + this.size) = value; }
            this.size++;
        }

        func operator[](int index) => int
        {
            raw { return *(this.data + index); }
        }

        func length() => int
        {
            return this.size;
        }

        // Map: takes a first-class function, returns a new array
        func map((int) => int transform) => DynamicArray
        {
            var result = DynamicArray(this.capacity);
            for (int i = 0; i < this.size; i++)
            {
                var val = this[i] |> transform;
                result.push(val);
            }
            return result;
        }

        // Grow: uses memcpy to resize the backing array
        private func grow() => void
        {
            var newCapacity = this.capacity * 2;
            raw
            {
                var newData = (int*)malloc(newCapacity * sizeof(int));
                memcpy((byte*)newData, (byte*)this.data, this.size * sizeof(int));
                free((byte*)this.data);
                this.data = newData;
            }
            this.capacity = newCapacity;
        }
    }

-------- Constraint Bounds [test_74_constraint_bounds.mingus] --------

  --- Source ---
module Test {
    extern func printf(string fmt, ...) => int;

    interface Printable {
        func print() => void;
    }

    interface HasSize {
        func size() => int;
    }

    class Message : Printable {
        string text;
        constructor(string t) { this.text = t; }
        func print() => void { printf("%s", this.text); }
    }

    class Container : Printable, HasSize {
        int count;
        constructor(int n) { this.count = n; }
        func print() => void { printf("Container(%d)", this.count); }
        func size() => int { return this.count; }
    }

    // Single constraint
    func printIt<T: Printable>(T* obj) => void {
        obj->print();
    }

    // Multiple constraints
    func printWithSize<T: Printable + HasSize>(T* obj) => void {
        printf("[size=%d] ", obj->size());
        obj->print();
    }

    // Unconstrained + constrained
    func wrapPrint<T, U: Printable>(T value, U* printer) => void {
        printf("wrap(%d): ", value);
        printer->print();
    }

    func main() => void {
        printf("=== Test 1: Single constraint ===\n");
        var msg = new Message("hello");
        printIt(msg);
        printf("\n");

        printf("=== Test 2: Multiple constraints ===\n");
        var c = new Container(5);
        printWithSize(c);
        printf("\n");

        printf("=== Test 3: Single constraint with different type ===\n");
        printIt(c);
        printf("\n");

        printf("=== Test 4: Mixed constrained and unconstrained ===\n");
        wrapPrint(42, msg);
        printf("\n");

        printf("=== Test 5: Turbofish with constraint ===\n");
        printIt::<Container>(c);
        printf("\n");

        delete msg;
        delete c;
    }
}
